/* ============================================================ * clinic-aggregate.js * 유튜브 상위노출 데이터를 "성형외과(병원) 단위"로 묶어 * 노출 점유율(%)을 계산하는 모듈. * * 사용 전제 (dashboard.html의 데이터 구조에 맞게 FIELD만 수정하세요) * 각 행(row) = 특정 키워드 검색결과에서 특정 영상이 차지한 1개 순위 * 예) { keyword: "코성형", rank: 3, videoId: "abc", * title: "...", description: "...", channelTitle: "..." } * ============================================================ */ /* ---------- 0. 필드명 매핑 (여기만 데이터에 맞게 고치면 됩니다) ---------- */ const FIELD = { keyword: 'keyword', rank: 'rank', title: 'title', description: 'description', channel: 'channelTitle', videoId: 'videoId', }; /* ---------- 1. 병원명 사전 ---------- * name : 대시보드에 표시할 정식 병원명 * patterns: 제목/설명/채널명에서 찾을 표기 변형들 (띄어쓰기 무시, 대소문자 무시) * 새 병원은 여기에 한 줄씩 추가하면 자동으로 집계에 잡힙니다. */ const CLINIC_PATTERNS = [ { name: '팝성형외과', patterns: ['팝성형외과', '팝성형', 'pop성형외과', 'popps'] }, { name: 'A성형외과', patterns: ['a성형외과', '에이성형외과'] }, { name: '뷰티라인성형외과', patterns: ['뷰티라인성형외과', '뷰티라인'] }, // TODO: 경쟁 병원 계속 추가 ]; /* 표기 흔들림 제거: 공백/특수문자 제거 + 소문자화 */ function normalize(text) { return String(text || '') .toLowerCase() .replace(/[\s\u200b·・.,\-_()\[\]]/g, ''); } /* 사전을 긴 패턴 우선으로 정렬 → "팝성형외과"가 "팝성형"보다 먼저 매칭됨 */ const SORTED_CLINICS = CLINIC_PATTERNS .map(c => ({ name: c.name, patterns: c.patterns.map(normalize) })) .sort((a, b) => Math.max(...b.patterns.map(p => p.length)) - Math.max(...a.patterns.map(p => p.length)) ); /* ---------- 2. 한 행에서 병원명 추출 ---------- * 채널명 > 제목 > 설명 순으로 신뢰. 못 찾으면 '기타'. */ function extractClinic(row) { const sources = [row[FIELD.channel], row[FIELD.title], row[FIELD.description]]; for (const src of sources) { const hay = normalize(src); if (!hay) continue; for (const clinic of SORTED_CLINICS) { if (clinic.patterns.some(p => p && hay.includes(p))) return clinic.name; } } return '기타'; } /* ---------- 3. 병원별 노출 점유율 집계 ---------- * options: * topN : 몇 위까지를 '노출'로 볼지 (기본 10) * weighted : true면 상위 순위에 가중치(1/rank)를 줘서 점유율 계산 * includeEtc : '기타'(미매칭)를 결과에 포함할지 * * 반환: [{ clinic, slots, score, share, keywords, avgRank, videos }] 점유율 내림차순 */ function aggregateByClinic(rows, options = {}) { const { topN = 10, weighted = false, includeEtc = false } = options; const inRange = rows.filter(r => Number(r[FIELD.rank]) >= 1 && Number(r[FIELD.rank]) <= topN); const weightOf = rank => (weighted ? 1 / Number(rank) : 1); const totalScore = inRange.reduce((sum, r) => sum + weightOf(r[FIELD.rank]), 0); const map = new Map(); for (const row of inRange) { const clinic = extractClinic(row); if (!map.has(clinic)) { map.set(clinic, { clinic, slots: 0, score: 0, rankSum: 0, keywords: new Set(), videos: [] }); } const acc = map.get(clinic); const rank = Number(row[FIELD.rank]); acc.slots += 1; acc.score += weightOf(rank); acc.rankSum += rank; acc.keywords.add(row[FIELD.keyword]); acc.videos.push(row); } return [...map.values()] .filter(a => includeEtc || a.clinic !== '기타') .map(a => ({ clinic: a.clinic, slots: a.slots, // 상위노출 차지 횟수 share: totalScore ? +(a.score / totalScore * 100).toFixed(1) : 0, // 노출 점유율 % keywords: a.keywords.size, // 노출된 키워드 수 avgRank: +(a.rankSum / a.slots).toFixed(1), // 평균 순위 videos: a.videos.sort((x, y) => x[FIELD.rank] - y[FIELD.rank]), })) .sort((a, b) => b.share - a.share || a.avgRank - b.avgRank); } /* ---------- 4. 사용 예시 ---------- * const ranking = aggregateByClinic(rawRows, { topN: 10, weighted: false }); * ranking.forEach((r, i) => { * console.log(`${i + 1}위 ${r.clinic} — 점유율 ${r.share}% / ${r.slots}건 / 평균 ${r.avgRank}위`); * }); * * 기존 표 렌더링 함수에서 "영상 1개 = 1행"을 "병원 1개 = 1행"으로 바꾸고, * 행 클릭 시 r.videos를 펼쳐 보여주면 드릴다운까지 됩니다. */ /* ---------- 5. 매칭 점검용 (초기 세팅 때 꼭 한 번) ---------- */ function auditUnmatched(rows) { const etc = rows.filter(r => extractClinic(r) === '기타'); const byChannel = {}; etc.forEach(r => { const c = r[FIELD.channel] || '(채널명 없음)'; byChannel[c] = (byChannel[c] || 0) + 1; }); return Object.entries(byChannel).sort((a, b) => b[1] - a[1]); // [채널명, 미매칭 건수] }
